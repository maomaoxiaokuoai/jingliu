/** Explicit own-account WebView session. No password interception, JS bridge, TLS override,
 * automatic login success, external browser cookie reading or injected authorization page. */
class EmbeddedLoginActivity:ComponentActivity() {
 companion object {private val leased=AtomicBoolean(false)}
 private var ownsLease=false
 private lateinit var platform:Platform
 private lateinit var graph:AppGraph
 private var web:WebView?=null
 private val token=CancelToken()
 private val visited=linkedSetOf<String>()
 private var status by mutableStateOf("准备官方登录页面")
 private var current by mutableStateOf("")
 private var ready by mutableStateOf(false)
 private var busy by mutableStateOf(false)
 private var desktop by mutableStateOf(false)
 private var screenshots by mutableStateOf(false)
 private var confirmSave by mutableStateOf(false)
 private var confirmScreenshot by mutableStateOf(false)
 private var pendingApp by mutableStateOf<String?>(null)
 private var savedRevision:String?=null
 override fun onCreate(savedInstanceState:Bundle?) {
  super.onCreate(savedInstanceState);enableEdgeToEdge()
  ownsLease=leased.compareAndSet(false,true);if(!ownsLease){finish();return}
  platform=runCatching{Platform.valueOf(intent.getStringExtra("platform").orEmpty())}.getOrNull()?:run{finish();return}
  if(!BrowserSessionPolicy.embeddedAllowed(platform)){finish();return}
  desktop=intent.getBooleanExtra("desktop",false);current=platform.loginUrl;graph=AppGraph.get(this)
  window.addFlags(WindowManager.LayoutParams.FLAG_SECURE)
  setContent {
   val settings by graph.settings.flow.collectAsState()
   CompositionLocalProvider(LocalUiSettings provides settings){LumaTheme(settings.appearance()) {
    BackHandler {if(web?.canGoBack()==true)web?.goBack()else finish()}
    Scaffold(topBar={TopAppBar(title={Column{Text(platform.title+if(desktop)" · 电脑版"else" · 手机版");Text(runCatching{Uri.parse(current).host}.getOrNull().orEmpty(),style=MaterialTheme.typography.labelSmall)}},
     navigationIcon={TextButton(onClick={finish()}){Text("关闭")}},actions={TextButton(onClick={desktop=!desktop;setAgent();web?.reload()}){Text(if(desktop)"切手机版"else"切电脑版")}})},
     bottomBar={Column(Modifier.navigationBarsPadding().padding(12.dp)) {
      Note(status)
      Row(Modifier.fillMaxWidth()) {
       TextButton(onClick={if(screenshots){screenshots=false;window.addFlags(WindowManager.LayoutParams.FLAG_SECURE)}else confirmScreenshot=true},modifier=Modifier.weight(1f)){Text(if(screenshots)"恢复截图保护"else"二维码截图…")}
       TextButton(onClick={openExternal()},modifier=Modifier.weight(1f)){Text("外部浏览器")}
      }
      Button(onClick={confirmSave=true},enabled=ready&&!busy,modifier=Modifier.fillMaxWidth()){Text(if(busy)"正在保存／校验…"else"我已登录，保存此页会话")}
     }}){padding->AndroidView(Modifier.padding(padding).fillMaxSize(),factory={createWebView().also{web=it;prepare(it)}})}
    if(confirmSave)AlertDialog(onDismissRequest={confirmSave=false},title={Text("保存本机会话？")},text={Text("读取的是镜流内置浏览器里属于 ${platform.title} 的 Cookie，不是外部浏览器或官方 App 的私有数据。会覆盖本机该平台的旧会话；Cookie 等同于凭据，只在本机加密保存。")},confirmButton={TextButton(onClick={confirmSave=false;saveSession()}){Text("确认保存")}},dismissButton={TextButton(onClick={confirmSave=false}){Text("取消")}})
    if(confirmScreenshot)AlertDialog(onDismissRequest={confirmScreenshot=false},title={Text("暂时允许截图？")},text={Text("仅用于保存此页面的登录二维码。页面密码或账号可能同时被截入；不要分享图片，注意相册云备份。关闭此浏览器后恢复保护。")},confirmButton={TextButton(onClick={confirmScreenshot=false;screenshots=true;window.clearFlags(WindowManager.LayoutParams.FLAG_SECURE)}){Text("允许本次截图")}},dismissButton={TextButton(onClick={confirmScreenshot=false}){Text("取消")}})
    pendingApp?.let{url->AlertDialog(onDismissRequest={pendingApp=null},title={Text("继续到官方 App？")},text={Text("这是官方页面请求的跳转，不代表授权已成功。返回后只有本页实际取得的 Cookie 才能保存。")},confirmButton={TextButton(onClick={pendingApp=null;launchApp(url)}){Text("打开")}},dismissButton={TextButton(onClick={pendingApp=null}){Text("留在页面")}})}
   }}
  }
 }
 @SuppressLint("SetJavaScriptEnabled")
 private fun createWebView()=WebView(this).apply {
  settings.javaScriptEnabled=true;settings.domStorageEnabled=true
  settings.allowFileAccess=false;settings.allowContentAccess=false
  settings.mixedContentMode=WebSettings.MIXED_CONTENT_NEVER_ALLOW;settings.safeBrowsingEnabled=true
  settings.setSupportMultipleWindows(false);settings.javaScriptCanOpenWindowsAutomatically=false
  settings.mediaPlaybackRequiresUserGesture=true
  CookieManager.getInstance().setAcceptCookie(true)
  CookieManager.getInstance().setAcceptThirdPartyCookies(this,false)
  settings.userAgentString=if(desktop)BrowserSessionPolicy.DESKTOP_AGENT else WebSettings.getDefaultUserAgent(this@EmbeddedLoginActivity)
  settings.useWideViewPort=desktop;settings.loadWithOverviewMode=desktop
  webViewClient=object:WebViewClient() {
   override fun shouldOverrideUrlLoading(view:WebView,request:WebResourceRequest):Boolean {
    if(!request.isForMainFrame)return false
    val url=request.url.toString()
    if(BrowserSessionPolicy.allowed(platform,url)){current=url;visited.add(url);return false}
    if(BrowserSessionPolicy.appScheme(platform,request.url.scheme.orEmpty())&&request.hasGesture()){pendingApp=url;return true}
    status="平台请求了外部或受限制的跳转。请用外部浏览器完成；镜流不会绕过站点登录限制。";return true
   }
   override fun onPageFinished(view:WebView,url:String) {if(BrowserSessionPolicy.allowed(platform,url)){current=url;visited.add(url);ready=true;status="在官方页面登录（可切电脑版查找二维码）。完成后由你确认保存；网页返回不等于已登录。"}}
   override fun onReceivedError(view:WebView,request:WebResourceRequest,error:WebResourceError){if(request.isForMainFrame)status="页面加载失败。可以重试或改用外部浏览器；旧账号未被清除。"}
  }
  webChromeClient=WebChromeClient()
  setDownloadListener{_,_,_,_,_->status="请返回解析页创建下载任务，网页登录页不执行文件下载。"}
 }
 private fun setAgent(){web?.settings?.apply{userAgentString=if(desktop)BrowserSessionPolicy.DESKTOP_AGENT else WebSettings.getDefaultUserAgent(this@EmbeddedLoginActivity);useWideViewPort=desktop;loadWithOverviewMode=desktop}}
 private fun prepare(view:WebView) {
  lifecycleScope.launch {
   try {val old=withContext(Dispatchers.IO){graph.load();graph.vault.get(platform)};savedRevision=old?.revision
    val cm=CookieManager.getInstance()
    cm.removeAllCookies { // isolate from the preceding platform's temporary WebView session
     if(token.cancelled)return@removeAllCookies
     WebStorage.getInstance().deleteAllData();view.clearCache(true)
     val cookies=old?.cookies.orEmpty().filter{platform.owns(it.domain)&&(it.expires==0L||it.expires>System.currentTimeMillis()/1000)}
     if(cookies.isEmpty()){view.loadUrl(current);return@removeAllCookies}
     var remaining=cookies.size
     cookies.forEach{c->cm.setCookie("https://${c.domain}${c.path}","${c.name}=${c.value}; Path=${c.path}; Secure${if(c.hostOnly)"" else "; Domain=${c.domain}"}${if(c.httpOnly)"; HttpOnly" else ""}"){
      remaining--;if(remaining==0&&!token.cancelled){cm.flush();view.loadUrl(current)}
     }}
    }
   }catch(_:Exception){status="安全存储或浏览器初始化失败，旧账号没有被修改。"}
  }
 }
 private fun saveSession() {
  if(busy||!ready)return
  val cm=CookieManager.getInstance()
  val headers=BrowserSessionPolicy.probes(platform,visited.toList()).associateWith{cm.getCookie(it).orEmpty()}
  busy=true
  lifecycleScope.launch {
   try {
    val result=withContext(Dispatchers.IO){
     val cookies=BrowserSessionPolicy.capture(platform,headers);require(cookies.isNotEmpty())
     var a=Account(platform,cookies)
     if(platform==Platform.BILIBILI)a=graph.bili.verify(a,token)
     token.check()
     synchronized(graph.vault){check(graph.vault.get(platform)?.revision==savedRevision){"当前账号已在别处修改，请重新打开登录页"};check(graph.vault.save(a,savedRevision))}
     a.stateLabel
    }
    setResult(Activity.RESULT_OK,Intent().putExtra("platform",platform.name).putExtra("status",result));finish()
   }catch(e:CancellationException){throw e}catch(_:Exception){status="保存或身份校验未通过。请确认页面已登录；没有覆盖旧账号，也没有生成假登录结果。"}
   finally{busy=false}
  }
 }
 private fun openExternal(){if(BrowserSessionPolicy.allowed(platform,current))runCatching{startActivity(Intent(Intent.ACTION_VIEW,Uri.parse(current)))}.onFailure{status="未找到浏览器"}}
 private fun launchApp(url:String){if(!BrowserSessionPolicy.appScheme(platform,Uri.parse(url).scheme.orEmpty()))return
  val pkg=NativeAppLauncher.installedPackage(this,platform)
  if(pkg==null){status="未检测到官方 App";return}
  runCatching{startActivity(Intent(Intent.ACTION_VIEW,Uri.parse(url)).setPackage(pkg).addCategory(Intent.CATEGORY_BROWSABLE))}.onFailure{status="官方 App 不接受此跳转，原网页登录会话保留。"}
 }
 override fun onDestroy(){token.cancel();if(ownsLease){web?.apply{stopLoading();clearHistory();clearCache(true);removeAllViews();destroy()};web=null;CookieManager.getInstance().removeAllCookies{leased.set(false)};WebStorage.getInstance().deleteAllData()};super.onDestroy()}
}
