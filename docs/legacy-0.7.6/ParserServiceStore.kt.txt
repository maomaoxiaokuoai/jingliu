package com.luma.downloader.engine

import android.content.Context
import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyProperties
import android.util.Base64
import com.luma.core.*
import com.luma.downloader.BuildConfig
import java.io.File
import java.security.KeyStore
import javax.crypto.Cipher
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.crypto.spec.GCMParameterSpec

data class ParserServiceConfig(val origin:String,val username:String="",val password:String="")
/** Separate encrypted settings; never shares SessionVault or serialises a platform account. */
class ParserServiceStore(context:Context) {
 private val file=File(context.noBackupFilesDir,"private-parser-v1.aes-gcm")
 private val alias="jingliu.private-parser.v1"
 @Synchronized fun read():ParserServiceConfig? {
  if(!file.exists())return null
  val e=Json.parse(file.readText());val cipher=Cipher.getInstance("AES/GCM/NoPadding")
  cipher.init(Cipher.DECRYPT_MODE,key(),GCMParameterSpec(128,Base64.decode(e.s("iv"),Base64.NO_WRAP)))
  val plain=cipher.doFinal(Base64.decode(e.s("data"),Base64.NO_WRAP))
  try {val v=Json.parse(plain.toString(Charsets.UTF_8));return ParserServiceConfig(v.s("origin"),v.s("username"),v.s("password"))}finally{plain.fill(0)}
 }
 @Synchronized fun write(value:ParserServiceConfig):ParserServiceConfig {
  val origin=PrivateParserPolicy.origin(value.origin,BuildConfig.DEBUG)
  PrivateParserPolicy.credentials(value.username,value.password)
  val result=value.copy(origin=origin)
  val cipher=Cipher.getInstance("AES/GCM/NoPadding");cipher.init(Cipher.ENCRYPT_MODE,key())
  val plain=Json.stringify(mapOf("origin" to origin,"username" to value.username,"password" to value.password)).toByteArray(Charsets.UTF_8)
  try {atomicText(file,Json.stringify(mapOf("iv" to Base64.encodeToString(cipher.iv,Base64.NO_WRAP),"data" to Base64.encodeToString(cipher.doFinal(plain),Base64.NO_WRAP))))}finally{plain.fill(0)}
  return result
 }
 fun parse(url:String,token:CancelToken):Media {
  val config=read()?:throw PlatformError("先配置自己的 parse-video-py 服务地址，未使用公共解析站","BACKUP_NOT_CONFIGURED")
  return PrivateParserClient(config.origin,config.username,config.password,BuildConfig.DEBUG).parse(url,token)
 }
 private fun key():SecretKey {
  val store=KeyStore.getInstance("AndroidKeyStore").apply{load(null)}
  (store.getKey(alias,null) as? SecretKey)?.let{return it}
  check(!file.exists()){"备用服务的加密密钥不可用，未读取或修改平台账号"}
  return KeyGenerator.getInstance(KeyProperties.KEY_ALGORITHM_AES,"AndroidKeyStore").apply {
   init(KeyGenParameterSpec.Builder(alias,KeyProperties.PURPOSE_ENCRYPT or KeyProperties.PURPOSE_DECRYPT).setBlockModes(KeyProperties.BLOCK_MODE_GCM).setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE).setKeySize(256).build())
  }.generateKey()
 }
}
